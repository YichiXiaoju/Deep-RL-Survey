\chapter{RL Fundamentals}
\section{Markov Decision Processes}
An MDP has states, actions, transition probabilities, rewards, a discount factor, and an initial distribution: $\mathcal M=(\States,\Actions,P,r,\gamma,\rho_0)$.
\begin{align}s_0&\sim\rho_0,&a_t&\sim\pi(\cdot\mid s_t),\\s_{t+1}&\sim P(\cdot\mid s_t,a_t).\end{align}
\section{Return, Value, Q, and Advantage}
\begin{align}
G_t&=\sum_{k=0}^{T-t-1}\gamma^k r_{t+k},\\
V^\pi(s)&=\E_\pi[G_t\mid s_t=s],\\
Q^\pi(s,a)&=\E_\pi[G_t\mid s_t=s,a_t=a],\\
A^\pi(s,a)&=Q^\pi(s,a)-V^\pi(s).
\end{align}
\begin{intuition}
$V(s)$: How good is this state if we follow the policy?\\
$Q(s,a)$: How good is taking this action first?\\
$A(s,a)$: Is the action better than average here?
\end{intuition}
\section{Bellman Equations}
For a fixed policy,
\begin{align}
V^\pi(s)&=\E_{a\sim\pi,s'\sim P}[r(s,a)+\gamma V^\pi(s')],\\
Q^\pi(s,a)&=\E_{s'\sim P}\left[r(s,a)+\gamma\E_{a'\sim\pi}Q^\pi(s',a')\right].
\end{align}
For optimal control,
\begin{equation}Q^*(s,a)=\E_{s'\sim P}\left[r(s,a)+\gamma\max_{a'}Q^*(s',a')\right].\end{equation}
The first equations evaluate a given policy. The last one looks for optimal actions.
\section{Monte Carlo and TD Learning}
\begin{align}
y_t^{\mathrm{MC}}&=G_t,\\
y_t^{\mathrm{TD}}&=r_t+\gamma(1-d_t)V_{\bar\phi}(s_{t+1}),\\
\delta_t&=y_t^{\mathrm{TD}}-V_\phi(s_t).
\end{align}
Here $d_t=1$ marks a truly terminal transition; time limits need care.
\begin{equation}\mathcal L_V(\phi)=\E[(V_\phi(s_t)-\operatorname{stopgrad}(y_t^{\mathrm{TD}}))^2].\end{equation}
\begin{pitfall}
A TD target is an estimate, not exact ground truth. We usually stop gradients through it.
\end{pitfall}
\section{Numerical Example}
If $r=2$, $\gamma=0.9$, $V(s')=5$, and $V(s)=8$, then $y^{\mathrm{TD}}=6.5$ and $\delta=-1.5$. The value prediction should move toward 6.5.
\section{Policy Gradient}
A common form is
\begin{equation}
\nabla_\theta J(\theta)=\E_\pi\left[\sum_t\nabla_\theta\log\pi_\theta(a_t\mid s_t)A^\pi(s_t,a_t)\right].
\end{equation}
Precise weighting depends on the objective convention. A positive advantage encourages an action.
\section{GAE}
GAE combines several TD residuals:
\begin{equation}\hat A_t^{\mathrm{GAE}(\gamma,\lambda)}=\sum_{l=0}^{T-t-1}(\gamma\lambda)^l\delta_{t+l}.\end{equation}
Handle episode boundaries correctly. A larger $\lambda$ often reduces bootstrap bias while increasing variance. PPO often computes advantages once per rollout and holds them fixed during several optimization passes.
